// 1) Get your cached NOTAMs
const raw = localStorage.getItem('fetchnotams:site:CZYZ');
const parsed = JSON.parse(raw);

// 2) Minimal DMS -> decimal helper for Q-line coords like 4459N07918W005
function dmsToDecimal(latDeg, latMin, ns, lonDeg, lonMin, ew) {
  const lat = (parseInt(latDeg,10) + parseInt(latMin,10)/60) * (ns === 'S' ? -1 : 1);
  const lon = (parseInt(lonDeg,10) + parseInt(lonMin,10)/60) * (ew === 'W' ? -1 : 1);
  return { lat, lon };
}

function extractFirstCoord(rawText) {
  if (!rawText) return null;
  const m = rawText.match(/\b(\d{2})(\d{2})([NS])(\d{3})(\d{2})([EW])\b/);
  if (!m) return null;
  return dmsToDecimal(m[1], m[2], m[3], m[4], m[5], m[6]);
}

// 3) Build records (cap to 120 markers)
const records = [];
const seen = new Set();

for (const item of (parsed.items || [])) {
  const coord = extractFirstCoord(item.raw);
  if (!coord) continue;
  const key = `${coord.lat.toFixed(6)},${coord.lon.toFixed(6)}`;
  if (seen.has(key)) continue;
  seen.add(key);
  records.push({
    lat: Number(coord.lat.toFixed(6)),
    lon: Number(coord.lon.toFixed(6)),
    start: item.startValidity || null,
    end: item.endValidity || null,
    raw: String(item.raw || "")
  });
  if (records.length >= 120) break;
}

// 4) Store payload + open map
const key = `fetchnotams:map:payload:${Date.now().toString(36)}-${Math.random().toString(36).slice(2,8)}`;
localStorage.setItem(key, JSON.stringify({ createdAt: Date.now(), records }));

const first = records[0];
const params = new URLSearchParams({ src: "fetchnotams", focus: "1", payload_key: key, lat: String(first.lat), lon: String(first.lon) });
window.open(`NotamsMap.html?${params.toString()}`, "_blank");
